cmake_minimum_required(VERSION 3.20)
project(metawork_windows_probe LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "The P0 probe requires a native Windows MSVC builder")
endif()
add_executable(metawork-windows-probe pipe-probe.cpp)
target_compile_features(metawork-windows-probe PRIVATE cxx_std_17)
target_compile_definitions(metawork-windows-probe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(metawork-windows-probe PRIVATE /W4 /WX /EHsc)
target_link_libraries(metawork-windows-probe PRIVATE advapi32)
set_property(TARGET metawork-windows-probe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
